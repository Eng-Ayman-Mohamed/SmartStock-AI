## Phase 6 — Dashboard Fixes Applied

### Fix 1: Inf MAPE Guard (CRITICAL)

**File:** `smartstock-backend/apps/forecasting/services.py`

```python
# Before:
if mape is not None:
    mape_pct = mape * 100 if mape <= 1.0 else mape
    confidence = max(0, min(100, round(100 - mape_pct)))

# After:
if mape is not None and not math.isinf(mape) and not math.isnan(mape):
    mape_pct = mape * 100 if mape <= 1.0 else mape
    confidence = max(0, min(100, round(100 - mape_pct)))
else:
    confidence = None
```

**DB Fix:** Fixed 1,015 inf MAPE values → NULL via raw SQL:
```sql
UPDATE forecasting_forecastresult
SET mape = NULL
WHERE mape = 'Infinity'::float OR mape = '-Infinity'::float;
```

### Fix 2: Frontend API Envelope Unwrap (HIGH)

**File:** `smartstock-frontend/src/features/dashboard/api.ts`

```typescript
function unwrap<T>(data: Record<string, unknown>): T {
  if (data && 'status' in data && 'data' in data) {
    return data.data as T;
  }
  return data as unknown as T;
}

export async function fetchLowStockItems(): Promise<ReorderAlert[]> {
  const { data } = await api.get('/inventory/stock-levels/low_stock/');
  return unwrap<ReorderAlert[]>(data);
}

export async function fetchAgentRuns(): Promise<AgentRun[]> {
  const { data } = await api.get('/audit/logs/agent-runs/', {
    params: { page_size: 100 },
  });
  return unwrap<AgentRun[]>(data);
}

export async function fetchPendingPOs(): Promise<PurchaseOrder[]> {
  const { data } = await api.get('/purchasing/orders/', {
    params: { status: 'pending_approval', page_size: 100 },
  });
  return unwrap<PurchaseOrder[]>(data);
}

export async function fetchSKUCount(): Promise<number> {
  const { data } = await api.get('/inventory/skus/', { params: { page_size: 1 } });
  if (data && typeof data === 'object' && 'meta' in data) {
    return (data.meta?.total as number) ?? 0;
  }
  if (data && typeof data === 'object' && 'count' in data) {
    return data.count as number;
  }
  return 0;
}
```

### Fix 4: Safe Float Renderer (HIGH)

**File:** `smartstock-backend/config/renderers.py`

```python
class SafeFloatJSONEncoder(json.JSONEncoder):
    """JSON encoder that converts inf/nan floats to None for JSON compliance."""

    def encode(self, o):
        return super().encode(self._sanitize(o))

    def _sanitize(self, o):
        if isinstance(o, float) and (math.isinf(o) or math.isnan(o)):
            return None
        if isinstance(o, dict):
            return {k: self._sanitize(v) for k, v in o.items()}
        if isinstance(o, (list, tuple)):
            return [self._sanitize(v) for v in o]
        return o
```
